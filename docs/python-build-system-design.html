1. 架构总览

将原来的三层架构（Starlark 规则 → Shell 脚本 → 系统命令）压缩为一层：Python 直接调用系统命令。

Before: 3 层 / 3 种语言
Starlark (.bzl)
  → 拼接 shell 脚本字符串
  → export 环境变量
  → source build.sh

Shell (.sh)
  → 读环境变量
  → 调 make / dd / parted
  → 设输出变量

Bazel action runner
  → 收集产物文件
After: 1 层 / 1 种语言
Python
  → 读 FINAL_CONFIG (dict)
  → subprocess.run(["make", ...])
  → subprocess.run(["dd", ...])
  → Path 对象管理产物

没有环境变量契约
没有 shell 脚本
没有 select()
没有 Starlark

2.3 条件解析

def resolve_conditions(config: dict, product: str, variant: str) -> dict:
    """解析条件标记，返回扁平的 FINAL_CONFIG。

    规则：
    - "packages"           → 无条件，保留
    - "+packages"          → 无条件追加到 "packages"
    - "packages:debug"     → variant 匹配时覆盖 "packages"
    - "+packages:debug"    → variant 匹配时追加到 "packages"
    - "+packages:gateway"  → product 匹配时追加到 "packages"
    - 条件不匹配的 key     → 丢弃

    先合并所有层（保留条件标记），最后统一解析。
    """
    result = {}
    active_conditions = {product, variant}

    for key, value in config.items():
        if key.startswith("+"):
            raw = key[1:]
            if ":" in raw:
                base_key, cond = raw.rsplit(":", 1)
                if cond in active_conditions:
                    result[base_key] = result.get(base_key, []) + value
            else:
                result[raw] = result.get(raw, []) + value
        elif ":" in key:
            base_key, cond = key.rsplit(":", 1)
            if cond in active_conditions:
                result[base_key] = value
        else:
            if key not in result:  # 无条件 key 不覆盖已追加的结果
                result[key] = value

    # 递归处理嵌套 dict
    for key, value in result.items():
        if isinstance(value, dict):
            result[key] = resolve_conditions(value, product, variant)

    return result

class BuildEngine:
    """构建引擎 — 管理依赖图、增量检查和 Docker 调度"""

    DEPENDENCY_GRAPH = {
        "kernel":     [],
        "bootloader": [],
        "rootfs":     [],
        "boot":       ["kernel"],
        "image":      ["boot", "bootloader", "rootfs"],
    }

    def __init__(self, config: dict):
        self.config = config
        self.docker = DockerRunner(config)
        self.source = SourceManager(config)
        self.cache = BuildCache(config)

    def build(self, target: str = "image"):
        """按拓扑序构建 target 及其全部依赖"""
        for component in self._topo_sort(target):
            if self.cache.is_up_to_date(component):
                log.info(f"  {component}: 无变更，跳过")
                continue

            log.info(f"  {component}: 开始构建...")
            builder = self._get_builder(component)
            builder.build(self.config)
            self.cache.store(component)

    def _get_builder(self, component: str) -> 'ComponentBuilder':
        """根据 platform 路由到对应的平台策略类"""
        platform = self.config["platform"]
        # e.g. "rockchip" → builder.platforms.rockchip.kernel.RockchipKernelBuilder
        module = import_module(f"builder.platforms.{platform}.{component}")
        return module.create_builder(self.docker, self.source)

4.2 核心基类: ComponentBuilder

ComponentBuilder — 组件构建基类（框架层） 
from abc import ABC, abstractmethod
from pathlib import Path
import subprocess

class ComponentBuilder(ABC):
    """组件构建基类 — 管理源码生命周期、补丁应用、产物收集。

    子类只需实现平台特有的 configure/compile/collect 方法。
    基类负责通用的构建框架：
      1. 确保源码就绪
      2. 重置源码树（保留编译产物实现增量编译）
      3. 应用补丁（平台补丁 → 板级补丁，按文件名排序）
      4. 调用子类的 configure → compile → collect
      5. 返回产物路径
    """

    component: str = ""     # 子类声明: "kernel", "bootloader", etc.

    def __init__(self, docker: 'DockerRunner', source: 'SourceManager'):
        self.docker = docker
        self.source = source

    def build(self, config: dict) -> dict:
        """执行完整的构建流程，返回产物路径 dict"""
        src_dir = self.source.ensure(self.component, config)

        if not config.get("_local_mode", {}).get(self.component):
            self.reset_source(src_dir)
            self.apply_patches(src_dir, config)

        self.configure(src_dir, config)
        self.compile(src_dir, config)
        return self.collect(src_dir, config)

    def reset_source(self, src_dir: Path):
        """重置源码树，保留 .o 等编译产物实现增量编译"""
        self.docker.run(["git", "checkout", "-f", "."], cwd=src_dir)

    def apply_patches(self, src_dir: Path, config: dict):
        """按序应用平台补丁 + 板级补丁"""
        platform = config["platform"]
        board = config["board"]

        platform_patches = sorted(
            Path(f"platform/{platform}/patches/{self.component}").glob("*.patch")
        )
        board_patches = sorted(
            Path(f"board/{board}/patches/{self.component}").glob("*.patch")
        )

        for patch in platform_patches + board_patches:
            self.docker.run(
                ["git", "apply", str(patch)],
                cwd=src_dir,
                check=True,
                fallback=["patch", "-p1", "-i", str(patch)]
            )

    @abstractmethod
    def configure(self, src_dir: Path, config: dict):
        """配置构建（如 make defconfig）"""
        ...

    @abstractmethod
    def compile(self, src_dir: Path, config: dict):
        """执行编译"""
        ...

    @abstractmethod
    def collect(self, src_dir: Path, config: dict) -> dict:
        """收集产物，返回 {name: Path} 映射"""
        ...

    # ── 工具方法 ──

    def make(self, src_dir: Path, targets: list, *,
             arch: str = "", cross: str = "", jobs: int = 0, extra: list = None):
        """封装 make 调用"""
        cmd = ["make"]
        if arch:
            cmd.append(f"ARCH={arch}")
        if cross:
            cmd.append(f"CROSS_COMPILE={cross}")
        if jobs:
            cmd.append(f"-j{jobs}")
        else:
            cmd.append(f"-j$(nproc)")
        cmd.extend(extra or [])
        cmd.extend(targets)
        self.docker.run(cmd, cwd=src_dir, check=True)

DockerRunner — Docker 容器执行封装 
class DockerRunner:
    """Docker 容器执行封装。

    所有构建命令通过 Docker 容器执行，确保环境一致性。
    rootfs 等需要 mount/chroot 的操作使用 privileged 模式。
    """

    def __init__(self, config: dict, project_dir: Path = None):
        self.project_dir = project_dir or Path.cwd()
        self.config = config

    def run(self, cmd: list, *, cwd: Path = None,
            env: dict = None, privileged: bool = False,
            check: bool = True, fallback: list = None) -> subprocess.CompletedProcess:
        """在 Docker 容器内执行命令"""
        docker_cmd = ["docker", "compose", "run", "--rm"]
        if privileged:
            docker_cmd.append("--privileged")
        if cwd:
            docker_cmd.extend(["-w", str(cwd)])
        if env:
            for k, v in env.items():
                docker_cmd.extend(["-e", f"{k}={v}"])
        docker_cmd.append("build")
        docker_cmd.extend(cmd)

        result = subprocess.run(docker_cmd, cwd=self.project_dir)
        if result.returncode != 0 and fallback:
            docker_cmd_fb = docker_cmd[:-len(cmd)] + fallback
            result = subprocess.run(docker_cmd_fb, cwd=self.project_dir, check=check)
        elif check and result.returncode != 0:
            raise BuildError(f"命令失败: {' '.join(cmd)}")
        return result

    def run_privileged(self, cmd: list, **kwargs):
        """特权模式执行（rootfs chroot/mount 操作需要）"""
        return self.run(cmd, privileged=True, **kwargs)
SourceManager — 源码仓库管理 
class SourceManager:
    """源码仓库管理 — 替代 Bazel module extensions + repository rules。

    功能：
    - 首次构建时 shallow clone 源码
    - 支持 commit 锁定（精确复现）
    - 支持 local_path 覆盖（本地开发模式）
    - 平台级共享仓库去重（如 rkbin）
    """

    def __init__(self, config: dict, sources_dir: Path = None):
        self.config = config
        self.sources_dir = sources_dir or Path("sources")

    def ensure(self, component: str, config: dict) -> Path:
        """确保源码就绪，返回源码目录路径"""
        comp_config = config.get(component, {})

        # 本地源码覆盖
        local = comp_config.get("local_path")
        if local:
            return Path(local)

        repo_dir = self.sources_dir / component / config["board"]

        if not repo_dir.exists():
            self._clone(
                repo=comp_config["repo"],
                branch=comp_config["branch"],
                dest=repo_dir,
                commit=comp_config.get("commit"),
            )
        elif comp_config.get("commit"):
            current = self._rev_parse(repo_dir)
            if current != comp_config["commit"]:
                self._fetch_checkout(repo_dir, comp_config["commit"])

        return repo_dir

    def _clone(self, repo, branch, dest, commit=None):
        dest.parent.mkdir(parents=True, exist_ok=True)
        cmd = ["git", "clone", "--depth=1", "-b", branch, repo, str(dest)]
        env = {"GIT_SSH_COMMAND": "ssh -o StrictHostKeyChecking=accept-new"}
        subprocess.run(cmd, env={**os.environ, **env},
                       check=True, timeout=1800)
        if commit:
            subprocess.run(["git", "fetch", "--depth=1", "origin", commit],
                           cwd=dest, check=True)
            subprocess.run(["git", "checkout", commit], cwd=dest, check=True)
BuildCache — 增量构建缓存 
class BuildCache:
    """增量构建缓存 — 基于内容哈希判断组件是否需要重建。

    哈希输入：
    - 组件配置值 (FINAL_CONFIG 中对应部分)
    - 源码 commit SHA
    - 补丁文件哈希
    - 构建脚本哈希 (平台策略类的源码)

    比 Bazel 的 blob 级缓存更精确 — 能区分"只改了 defconfig"
    和"源码 commit 变了"。
    """

    def __init__(self, config: dict, cache_dir: Path = None):
        self.config = config
        self.board = config["board"]
        self.product = config.get("product", "default")
        self.variant = config.get("variant", "release")
        self.target_dir = Path("target") / self.board / self.product / self.variant

    def is_up_to_date(self, component: str) -> bool:
        hash_file = self.target_dir / component / ".build_hash"
        if not hash_file.exists():
            return False
        stored = hash_file.read_text().strip()
        return stored == self._compute_hash(component)

    def store(self, component: str):
        hash_file = self.target_dir / component / ".build_hash"
        hash_file.parent.mkdir(parents=True, exist_ok=True)
        hash_file.write_text(self._compute_hash(component))

    def _compute_hash(self, component: str) -> str:
        import hashlib
        h = hashlib.sha256()
        # 配置值
        h.update(json.dumps(self.config.get(component, {}),
                            sort_keys=True).encode())
        # 源码 commit
        src_dir = Path("sources") / component / self.board
        if src_dir.exists():
            commit = subprocess.run(
                ["git", "rev-parse", "HEAD"], cwd=src_dir,
                capture_output=True, text=True
            ).stdout.strip()
            h.update(commit.encode())
        # 补丁哈希
        for patch_dir in [
            Path(f"platform/{self.config['platform']}/patches/{component}"),
            Path(f"board/{self.board}/patches/{component}"),
        ]:
            if patch_dir.exists():
                for p in sorted(patch_dir.glob("*.patch")):
                    h.update(p.read_bytes())
        return h.hexdigest()[:16]
ChrootContext — 安全的 chroot 上下文管理器 
class ChrootContext:
    """Chroot 上下文管理器 — 自动管理 mount/umount 生命周期。

    替代 shell 脚本中手动 mount + 祈祷 cleanup 不漏的模式。
    无论构建成功或失败，__exit__ 都会清理所有挂载点。
    """

    def __init__(self, rootfs_dir: Path, docker: DockerRunner):
        self.rootfs = rootfs_dir
        self.docker = docker
        self._mounts = []  # 按挂载顺序记录，反序 umount

    def __enter__(self):
        self._mount("proc", self.rootfs / "proc", fstype="proc")
        self._mount("sysfs", self.rootfs / "sys", fstype="sysfs")
        self._bind("/dev", self.rootfs / "dev")
        self._bind("/dev/pts", self.rootfs / "dev/pts")
        return self

    def __exit__(self, *exc):
        # 反序 umount，确保不漏
        for mount_point in reversed(self._mounts):
            self.docker.run_privileged(
                ["umount", "-l", str(mount_point)], check=False)
        return False

    def run(self, cmd: list, **kwargs):
        """在 chroot 内执行命令"""
        self.docker.run_privileged(
            ["chroot", str(self.rootfs)] + cmd, **kwargs)

    def bind_mount(self, src: str, dest: Path = None):
        """绑定挂载（如 APT cache）"""
        dest = dest or self.rootfs / src.lstrip("/")
        self._bind(src, dest)

    def _mount(self, src, dest, fstype=None):
        dest.mkdir(parents=True, exist_ok=True)
        cmd = ["mount"]
        if fstype:
            cmd.extend(["-t", fstype])
        cmd.extend([src, str(dest)])
        self.docker.run_privileged(cmd, check=True)
        self._mounts.append(dest)

    def _bind(self, src, dest):
        dest.mkdir(parents=True, exist_ok=True)
        self.docker.run_privileged(
            ["mount", "-o", "bind", str(src), str(dest)], check=True)
        self._mounts.append(dest)

RockchipBootloaderBuilder — 替代 bootloader/rockchip/build.sh 
class RockchipBootloaderBuilder(BootloaderBuilder):
    """Rockchip 平台 Bootloader 构建策略。

    原 bootloader/rockchip/build.sh 的完整替代。
    INI 解析从 grep/awk/sed/cut 链替换为 configparser。
    """

    ARCH = "arm"  # U-Boot 用 arm 不是 arm64
    CROSS = "aarch64-linux-gnu-"

    def configure(self, src_dir: Path, config: dict):
        self.make(src_dir, [config["bootloader"]["defconfig"]],
                  arch=self.ARCH, cross=self.CROSS)

    def compile(self, src_dir: Path, config: dict):
        firmware_dir = self.source.ensure_firmware("rockchip", config)
        ini_prefix = config["rkbin"]["ini_prefix"]
        trust_prefix = config["rkbin"].get("trust_ini_prefix", ini_prefix)

        # 解析 RKTRUST INI — 提取 BL31/BL32 路径
        trust_ini = firmware_dir / "RKTRUST" / f"{trust_prefix}TRUST.ini"
        bl31, bl32 = self._parse_trust_ini(trust_ini, firmware_dir)

        # 复制固件到 U-Boot 源码目录
        shutil.copy(bl31, src_dir / "bl31.elf")
        if bl32:
            shutil.copy(bl32, src_dir / "tee.bin")

        # 编译 U-Boot → u-boot.itb (FIT 格式)
        extra = [f"BL31={src_dir}/bl31.elf"]
        if bl32:
            extra.append(f"TEE={src_dir}/tee.bin")
        jobs = config.get("jobs", 0)
        self.make(src_dir, ["u-boot.itb"],
                  arch=self.ARCH, cross=self.CROSS, jobs=jobs, extra=extra)

        # 解析 RKBOOT INI — 生成 idbloader.img
        loader_ini = firmware_dir / "RKBOOT" / f"{ini_prefix}MINIALL.ini"
        ddr_bin, spl_bin = self._parse_loader_ini(loader_ini, firmware_dir)
        self.docker.run([
            str(src_dir / "tools/mkimage"), "-n", "rk3568", "-T", "rksd",
            "-d", f"{ddr_bin}:{spl_bin}", str(src_dir / "idbloader.img")
        ])

        # 生成 miniloader.bin (用于 upgrade_tool 刷写)
        self.docker.run([
            str(firmware_dir / "tools/boot_merger"), str(loader_ini)
        ], cwd=firmware_dir)

    def _parse_trust_ini(self, ini_path: Path, fw_dir: Path):
        """用 configparser 解析 RKTRUST INI，替代 shell 的 grep|cut 链"""
        import configparser
        cfg = configparser.ConfigParser()
        cfg.read(ini_path)
        bl31 = fw_dir / cfg.get("BL31_OPTION", "PATH", fallback="")
        bl32_path = cfg.get("BL32_OPTION", "PATH", fallback="")
        bl32 = (fw_dir / bl32_path) if bl32_path else None
        return bl31, bl32

    def collect(self, src_dir: Path, config: dict) -> dict:
        firmware_dir = self.source.ensure_firmware("rockchip", config)
        ini_prefix = config["rkbin"]["ini_prefix"]
        return {
            "bootloader": src_dir / "u-boot.itb",
            "idbloader":  src_dir / "idbloader.img",
            "miniloader": firmware_dir / f"{ini_prefix}_loader_v*.bin",  # glob
        }
RockchipRootfsBuilder — 替代 rootfs/rockchip/build_base.sh + build_customize.sh 
class RockchipRootfsBuilder(RootfsBuilder):
    """Rockchip 平台 Rootfs 构建策略。

    合并了原来的 build_base.sh 和 build_customize.sh 两个脚本。
    ChrootContext 自动管理 mount/umount 生命周期。
    """

    def compile(self, src_dir: Path, config: dict):
        rootfs_dir = self._work_dir / "rootfs"
        rootfs_dir.mkdir(parents=True, exist_ok=True)

        tarball = self.source.ensure_rootfs_tarball(config)

        # Phase 1: Base rootfs
        self.docker.run_privileged(
            ["tar", "xf", str(tarball), "-C", str(rootfs_dir)])
        shutil.copy("/usr/bin/qemu-aarch64-static",
                     rootfs_dir / "usr/bin/")

        with ChrootContext(rootfs_dir, self.docker) as chroot:
            # 挂载 APT 缓存（持久化，跨构建共享）
            apt_cache = rootfs_dir / "var/cache/apt/archives"
            apt_cache.mkdir(parents=True, exist_ok=True)
            chroot.bind_mount("/cache/apt", apt_cache)

            chroot.run(["apt-get", "update"])
            chroot.run(["apt-get", "install", "-y",
                         "--no-install-recommends"]
                        + config["rootfs"]["packages"])
            chroot.run(["apt-get", "clean"])

        # Phase 2: Customize (overlay + custom debs)
        board = config["board"]
        overlay_dir = Path(f"board/{board}/overlay")
        if overlay_dir.exists():
            self.docker.run_privileged(
                ["cp", "-a", f"{overlay_dir}/.", str(rootfs_dir)])

        for deb in config["rootfs"].get("custom_debs", []):
            with ChrootContext(rootfs_dir, self.docker) as chroot:
                chroot.run(["dpkg", "-i", f"/tmp/{deb}"])

        # 压缩
        output = self._work_dir / "rootfs.tar.gz"
        self.docker.run_privileged(
            ["tar", "-czf", str(output), "-C", str(rootfs_dir), "."])

    def collect(self, src_dir: Path, config: dict) -> dict:
        return {"rootfs": self._work_dir / "rootfs.tar.gz"}
RockchipImageBuilder — 替代 image/rockchip/build_boot.sh + build_image.sh 
class RockchipImageBuilder(ImageBuilder):
    """Rockchip 平台镜像组装策略。

    合并了原来的 build_boot.sh 和 build_image.sh。
    分区偏移计算从 shell 算术替换为 Python int()。
    """

    SECTOR_SIZE = 512
    IDBLOADER_SECTOR = 64
    ROOTFS_PARTUUID = "614e0000-0000-4000-8000-000000000000"

    def compile(self, src_dir: Path, config: dict):
        # 获取依赖组件的产物
        kernel_outputs = self._deps["kernel"]
        bootloader_outputs = self._deps["bootloader"]
        rootfs_outputs = self._deps["rootfs"]

        partitions = config["partitions"]

        # Phase 1: 构建 boot 分区
        boot_img = self._build_boot_partition(kernel_outputs, config)

        # Phase 2: 生成分区配置文件
        partition_file = self._generate_partition_config(partitions, config)

        # Phase 3: 组装磁盘镜像
        raw_img = self._assemble_disk_image(
            boot_img, bootloader_outputs, rootfs_outputs,
            partitions, config)

        self._raw_img = raw_img

    def _build_boot_partition(self, kernel, config):
        """构建 ext4 boot 分区: Image + DTB + DTBO + extlinux.conf"""
        boot_dir = self._work_dir / "boot"
        boot_dir.mkdir(parents=True, exist_ok=True)

        # 复制内核和设备树
        shutil.copy(kernel["image"], boot_dir / "Image")
        dtb_dir = boot_dir / "dtb"
        dtb_dir.mkdir(exist_ok=True)
        shutil.copy(kernel["dtb"], dtb_dir)

        # 生成 extlinux.conf
        self._generate_extlinux(boot_dir, config)

        # 创建 ext4 镜像
        boot_img = self._work_dir / "boot.img"
        size_mb = config.get("boot", {}).get("size_mb", 256)
        self.docker.run_privileged([
            "dd", "if=/dev/zero", f"of={boot_img}",
            "bs=1M", f"count={size_mb}"
        ])
        self.docker.run_privileged([
            "mkfs.ext4", "-F", "-L", "boot", str(boot_img)
        ])
        # mount + copy + umount
        ...
        return boot_img

    def _assemble_disk_image(self, boot, bootloader, rootfs, partitions, config):
        """组装 GPT 磁盘镜像"""
        raw_img = self._work_dir / "raw.img"

        # 计算镜像大小
        total_sectors = self._calculate_image_size(partitions)
        total_bytes = total_sectors * self.SECTOR_SIZE
        self.docker.run_privileged([
            "dd", "if=/dev/zero", f"of={raw_img}",
            "bs=1", "count=0", f"seek={total_bytes}"
        ])

        # GPT 分区表
        self.docker.run_privileged([
            "parted", "-s", str(raw_img), "mklabel", "gpt"
        ])
        for entry in partitions["entries"]:
            if entry["type"] != "raw":
                self.docker.run_privileged([
                    "parted", "-s", str(raw_img), "mkpart",
                    entry["name"], entry["type"],
                    f"{entry['_start_bytes']}B",
                    f"{entry['_end_bytes']}B"
                ])

        # 写入 idbloader (sector 64)
        self.docker.run_privileged([
            "dd", f"if={bootloader['idbloader']}", f"of={raw_img}",
            f"seek={self.IDBLOADER_SECTOR}", "conv=notrunc"
        ])

        # 写入 bootloader + boot + rootfs ...
        return raw_img

    def collect(self, src_dir: Path, config: dict) -> dict:
        return {"image": self._raw_img}

6.2 中间格式定义

@dataclass
class Partition:
    name: str
    size: str           # "4M", "256M", "8G", "remaining"
    type: str           # "raw", "ext4", "fat32"
    offset: str = ""    # 可选, "32K", "0x4000"
    image: str = ""     # 对应的构建产物: "bootloader.idbloader"

@dataclass
class PartitionTable:
    format: str             # "gpt" | "mbr"
    sector_size: int = 512
    entries: list[Partition] = field(default_factory=list)

class RockchipPartitionConverter:
    """Rockchip parameter.txt 生成器"""

    def convert(self, table: PartitionTable) -> str:
        mtdparts = []
        for entry in table.entries:
            size_sectors = self._to_sectors(entry.size)
            offset_sectors = self._to_sectors(entry.offset) if entry.offset else "-"
            name = entry.name
            if entry.size == "remaining":
                mtdparts.append(f"-@{offset_sectors:#010x}({name}:grow)")
            else:
                mtdparts.append(f"{size_sectors:#010x}@{offset_sectors:#010x}({name})")
        return f"CMDLINE:mtdparts=rk29xxnand:{','.join(mtdparts)}"

7. Flash 脚本生成

class FlashGenerator:
    """从 FINAL_CONFIG 自动生成 flash.sh"""

    def generate(self, config: dict, target_dir: Path):
        flash_tool = config["flash_tool"]
        board = config["board"]
        product = config.get("product", "default")
        variant = config.get("variant", "release")

        script = f"""#!/bin/bash
# 自动生成 — 请勿手动编辑
# board: {board}  product: {product}  variant: {variant}
set -euo pipefail
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
FLASH_TOOL="{flash_tool}"

flash_bootloader() {{
    echo "刷写 bootloader..."
    $FLASH_TOOL db "$SCRIPT_DIR/bootloader/miniloader.bin"
    sleep 1
    $FLASH_TOOL wl 0x40 "$SCRIPT_DIR/bootloader/idbloader.img"
    $FLASH_TOOL wl 0x4000 "$SCRIPT_DIR/bootloader/bootloader.img"
}}

flash_kernel() {{
    echo "刷写内核..."
    $FLASH_TOOL wl 0x8000 "$SCRIPT_DIR/boot/boot.img"
}}

flash_rootfs() {{
    echo "刷写 rootfs..."
    $FLASH_TOOL wl 0x40000 "$SCRIPT_DIR/rootfs/rootfs.img"
}}

flash_all() {{
    flash_bootloader
    flash_kernel
    flash_rootfs
    echo "刷写完成，重启设备..."
    $FLASH_TOOL rd
}}

case "${{1:-all}}" in
    bootloader) flash_bootloader ;;
    kernel)     flash_kernel ;;
    rootfs)     flash_rootfs ;;
    all)        flash_all ;;
    *) echo "用法: $0 [bootloader|kernel|rootfs|all]" ;;
esac
"""
        flash_sh = target_dir / "flash.sh"
        flash_sh.write_text(script)
        flash_sh.chmod(0o755)

  • select()
  • config_setting
  • --define=
  • --config=
  • build_setting
  • 环境变量契约
  • Loading/Analysis 两阶段
  • Starlark 语言
  • BUILD.bazel 文件

保留

  • 三层配置继承
  • 框架/策略分离
  • Docker 构建环境
  • 补丁两级归属
  • 增量编译 (git checkout -f)
  • APT 缓存持久化
  • SSH 密钥注入
  • 产物收集机制

新增

  • product 抽象
  • variant (debug/release)
  • 条件标记 + 追加语义
  • 分区表中间格式
  • flash.sh 自动生成
  • 内容哈希增量
  • .flange/ 持久化
  • lunch 完整格式
  • ChrootContext 安全管理
